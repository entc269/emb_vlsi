
<!DOCTYPE html>
<html>
<head>

    <title>Simple Calculator</title>

    <script>

        var expression = "";

        function press(value) {

            expression += value;

            document.getElementById("display").value = expression;

        }

        function calculate() {

            try {

                if (expression == "") {

                    alert("Please enter a calculation.");

                    return;
                }

                if (!/^[0-9+\-*/.() ]+$/.test(expression)) {

                    alert("Invalid calculation.");

                    return;
                }

                var result = Function(
                    '"use strict"; return (' + expression + ')'
                )();

                if (!Number.isFinite(result)) {

                    alert("Cannot divide by zero.");

                    return;
                }

                document.getElementById("display").value =
                    expression + " = " + result;

                expression = String(result);

            } catch (error) {

                alert("Invalid calculation.");

            }

        }

        function square() {

            if (expression == "" || isNaN(Number(expression))) {

                alert("Enter a number first.");

                return;
            }

            var original = expression;

            var result = Number(expression) * Number(expression);

            document.getElementById("display").value =
                original + "² = " + result;

            expression = String(result);

        }

        function clearDisplay() {

            expression = "";

            document.getElementById("display").value = "";

        }

        function backspace() {

            expression = expression.slice(0, -1);

            document.getElementById("display").value = expression;

        }

    </script>

</head>

<body>

    <h2>Simple Calculator</h2>

    <input type="text" id="display" readonly size="30">

    <button onclick="press('*')">*</button>

    <br><br>

    <button onclick="press('0')">0</button>
    <button onclick="press('.')">.</button>
    <button onclick="press('/')">/</button>
    <button onclick="calculate()">=</button>

    <br><br>

    <button onclick="square()">Square</button>
    <button onclick="backspace()">Delete</button>
    <button onclick="clearDisplay()">Clear</button>

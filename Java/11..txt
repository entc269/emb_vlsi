    <div class="calculator">

        <h2>JavaScript Calculator</h2>

        <input type="number" id="num1"
            placeholder="Enter first number" step="any">

        <input type="number" id="num2"
            placeholder="Enter second number" step="any">

        <div class="buttons">

            <button onclick="calculate('+')">+</button>

            <button onclick="calculate('-')">-</button>

            <button onclick="calculate('*')">*</button>

            <button onclick="calculate('/')">/</button>

            <button onclick="square()">Square</button>

            <button onclick="clearAll()">Clear</button>

        </div>

        <h3>Result</h3>

        <input type="text" id="result"
            placeholder="Result" readonly>

    </div>

    <script>

        function calculate(operator) {

            let first = document.getElementById("num1").value;

            let second = document.getElementById("num2").value;

            if (first.trim() === "" || second.trim() === "") {

                alert("Please enter both numbers!");

                return;
            }

            let a = Number(first);

            let b = Number(second);

            if (!Number.isFinite(a) || !Number.isFinite(b)) {

                alert("Please enter valid numbers!");

                return;
            }

            let answer;

            switch (operator) {

                case "+":

                    answer = a + b;

                    break;

                case "-":

                    answer = a - b;

                    break;

                case "*":

                    answer = a * b;

                    break;

                case "/":

                    if (b === 0) {

                        alert("Cannot divide by zero!");

                        return;
                    }

                    answer = a / b;

                    break;

            }

            document.getElementById("result").value = answer;

        }

        function square() {

            let value = document.getElementById("num1").value;

            if (value.trim() === "") {

                alert("Please enter the first number!");

                return;
            }

            let number = Number(value);

            if (!Number.isFinite(number)) {

                alert("Please enter a valid number!");

                return;
            }

            document.getElementById("result").value =
                number * number;

        }

        function clearAll() {

            document.getElementById("num1").value = "";

            document.getElementById("num2").value = "";

            document.getElementById("result").value = "";

        }

    </script>

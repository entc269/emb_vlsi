
<!DOCTYPE html>

    <title>Multiplication Table</title>
</head>

<body>

<script>

    // Accept input
    let num = parseInt(prompt("Enter a number:"), 10);

    if (isNaN(num)) {
        document.write("Please enter a valid number.");
    }
    else {

        // FOR LOOP
        document.write("<h3>Using FOR Loop</h3>");

        for (let i = 1; i <= 10; i++) {
            document.write(
                num + " x " + i + " = " + (num * i) + "<br>"
            );
        }

        // WHILE LOOP
        document.write("<h3>Using WHILE Loop</h3>");

        let j = 1;

        while (j <= 10) {
            document.write(
                num + " x " + j + " = " + (num * j) + "<br>"
            );

            j++;
        }

        // DO-WHILE LOOP
        document.write("<h3>Using DO-WHILE Loop</h3>");

        let k = 1;

        do {
            document.write(
                num + " x " + k + " = " + (num * k) + "<br>"
            );

            k++;

        } while (k <= 10);
    }

</script>

    <script>

        function showError(id, message) {

            var field = document.getElementById(id);

            var error = document.getElementById(id + "Error");

            var value = field.value.trim();

            if (value === "") {

                error.textContent = message;

                return false;
            }

            if (id === "name" && !/^[A-Za-z ]+$/.test(value)) {

                error.textContent = "Enter letters and spaces only.";

                return false;
            }

            if (id === "mobile" && !/^[6-9][0-9]{9}$/.test(value)) {

                error.textContent =
                    "Enter a valid 10-digit Indian mobile number.";

                return false;
            }

            if (id === "email" &&
                !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value)) {

                error.textContent = "Enter a valid email address.";

                return false;
            }

            error.textContent = "";

            return true;
        }


        function validateGender() {

            var gender = document.getElementById("gender");

            var error = document.getElementById("genderError");

            if (gender.value === "") {

                error.textContent = "Please select your gender.";

                return false;
            }

            error.textContent = "";

            return true;
        }


        function setupValidation() {

            var fields = [

                ["name", "Please enter your name."],

                ["address", "Please enter your address."],

                ["city", "Please enter your city."],

                ["state", "Please enter your state."],

                ["mobile", "Please enter your mobile number."],

                ["email", "Please enter your email ID."]

            ];

            fields.forEach(function(item) {

                var id = item[0];

                var message = item[1];

                document.getElementById(id).addEventListener(
                    "input", function() {

                        showError(id, message);

                    }
                );

            });

            document.getElementById("gender").addEventListener(
                "change", validateGender
            );

        }


        window.addEventListener("DOMContentLoaded", setupValidation);


        function validateForm() {

            var valid = true;

            if (!showError("name", "Please enter your name."))
                valid = false;

            if (!showError("address", "Please enter your address."))
                valid = false;

            if (!showError("city", "Please enter your city."))
                valid = false;

            if (!showError("state", "Please enter your state."))
                valid = false;

            if (!validateGender())
                valid = false;

            if (!showError("mobile", "Please enter your mobile number."))
                valid = false;

            if (!showError("email", "Please enter your email ID."))
                valid = false;

            if (!valid) return false;


            var name = document.getElementById("name").value.trim();

            var address = document.getElementById("address").value.trim();

            var city = document.getElementById("city").value.trim();

            var state = document.getElementById("state").value.trim();

            var gender = document.getElementById("gender").value;

            var mobile = document.getElementById("mobile").value.trim();

            var email = document.getElementById("email").value.trim();


            document.getElementById("welcomeName").textContent = name;

            document.getElementById("welcomeNameDisplay").textContent = name;

            document.getElementById("welcomeAddress").textContent = address;

            document.getElementById("welcomeCity").textContent = city;

            document.getElementById("welcomeState").textContent = state;

            document.getElementById("welcomeGender").textContent = gender;

            document.getElementById("welcomeMobile").textContent = mobile;

            document.getElementById("welcomeEmail").textContent = email;


            document.getElementById("formPage").hidden = true;

            document.getElementById("welcomePage").hidden = false;

            return false;
        }


        function resetErrors() {

            var errors = document.querySelectorAll(".errorMessage");

            errors.forEach(function(error) {

                error.textContent = "";

            });

        }

    </script>

<script>

function calculateSets() {

    // Read input and create sets
    let A = new Set(
        document.getElementById("courseA").value
        .split(",")
        .map(x => x.trim())
        .filter(x => x !== "")
    );

    let B = new Set(
        document.getElementById("courseB").value
        .split(",")
        .map(x => x.trim())
        .filter(x => x !== "")
    );

    if (A.size === 0 || B.size === 0) {

        alert("Please enter courses for both students!");

        return;
    }

    // Union
    let union = new Set([...A, ...B]);

    // Intersection
    let intersection = new Set(
        [...A].filter(x => B.has(x))
    );

    // Difference
    let difference = new Set(
        [...A].filter(x => !B.has(x))
    );

    // Symmetric Difference
    let symmetric = new Set(
        [...union].filter(x => !(A.has(x) && B.has(x)))
    );

    // Display results
    document.getElementById("output").innerHTML = `

        <h3>Course Registration Results</h3>

        <b>Student A:</b> ${[...A].join(", ")}<br><br>

        <b>Student B:</b> ${[...B].join(", ")}<br><br>

        <b>Union:</b> ${[...union].join(", ")}<br><br>

        <b>Intersection:</b>
        ${[...intersection].join(", ") || "None"}<br><br>

        <b>Difference (A - B):</b>
        ${[...difference].join(", ") || "None"}<br><br>

        <b>Symmetric Difference:</b>
        ${[...symmetric].join(", ") || "None"}

    `;

}

</script>
